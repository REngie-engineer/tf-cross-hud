event HudHealthChanged
{
    Animate HealthBar "wide" "player_health" Linear 0.1
}

event HudAmmoChanged
{
    Animate ClipBar "wide" "player_clip" Linear 0.1
    Animate ReserveBar "wide" "player_ammo" Linear 0.1
}

event HudMetalChanged
{
    Animate MetalBar "wide" "player_metal" Linear 0.1
}

event HudSpyCloakChanged
{
    Animate CloakBar "tall" "spy_cloak_level" Linear 0.1
    SetVisible CloakBar "1"
}

event HudMedicChargeChanged
{
    Animate ChargeBar "tall" "medic_charge_level" Linear 0.1
    SetVisible ChargeBar "1"
}

event HudMinModeOn
{
    SetVisible HealthBar "1"
    SetVisible ClipBar "1"
    SetVisible ReserveBar "1"
    SetVisible MetalBar "1"
    SetVisible CloakBar "1"
    SetVisible ChargeBar "1"
}

event HudMinModeOff
{
    SetVisible HealthBar "1"
    SetVisible ClipBar "1"
    SetVisible ReserveBar "1"
    SetVisible MetalBar "1"
    SetVisible CloakBar "1"
    SetVisible ChargeBar "1"
}
